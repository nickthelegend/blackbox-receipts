STATIC .github/workflows/attest-diff.yml:
# Receipt-gated merge on this repo's pull requests. When a pull request carries an agent's exported
# session (receipts/<session>/session.jsonl, written by `blackbox-receipts export`), every other file it
# changes must be exactly what a signed write_file step in that session produced: a human edit after the
# agent's write, or a file written around the proxy, fails the job and is named in the JUnit report.
# Pull requests without an exported session are not agent changes; the job says so and passes, unless the
# branch is an agent branch (blackbox/*). The gate is opt-in: it covers only requests that carry an export.
name: attest-diff

on:
  pull_request:

permissions:
  contents: read

jobs:
  attest-diff:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      # The gate runs from the base commit, never from the code under review: a pull request that edits
      # checkDiff, the CLI or attest-diff-pr.sh can't certify itself. The base is checked out in its own
      # worktree, installed and built there, and its copy of the script gates this checkout.
      - name: Build the gate from the base commit
        run: |
          GATE="$RUNNER_TEMP/gate"
          git worktree add --detach "$GATE" "${{ github.event.pull_request.base.sha }}"
          cd "$GATE" && pnpm install --frozen-lockfile && pnpm --filter @blackbox/verify build && pnpm --filter @blackbox/receipts build && pnpm --filter @blackbox/meterless build && pnpm --filter @blackbox/proxy build
      - name: Every changed file must come from a signed write_file step
        env:
          # Repository variable (PEM): an export signed by any other proxy key fails. Unset, the gate fails
          # closed with "Setup required" unless the repository variable BLACKBOX_REQUIRE_PINNED_KEY is 0
          # (then the export's own key is trusted and the result says UNPINNED).
          BLACKBOX_PUBLIC_KEY: ${{ vars.BLACKBOX_PUBLIC_KEY }}
          BLACKBOX_REQUIRE_PINNED_KEY: ${{ vars.BLACKBOX_REQUIRE_PINNED_KEY || '1' }}
          # An agent branch (blackbox/*) without an export is red instead of NOT APPLICABLE.
          BLACKBOX_AGENT_BRANCH: ${{ github.head_ref }}
        run: |
          GATE="$RUNNER_TEMP/gate"
          bash "$GATE/scripts/ci/attest-diff-pr.sh" "${{ github.event.pull_request.base.sha }}" attest-diff
      - if: always()
        uses: actions/upload-artifact@v4
        with:
          name: attest-diff
          path: attest-diff/
          if-no-files-found: ignore

# On a merge request, both export checks run a verifier built from the MR's base commit in its own
# worktree (like verify:attest-diff), so an MR that edits packages/verify or the CLI can't green its own
# receipts. Branch pipelines (the default branch after merge) use this pipeline's build.
.verifier-from-base: &verifier-from-base
  - |
    BLACKBOX_CLI=packages/proxy/dist/cli.js
    SCRIPTS=scripts/ci
    if [ "$CI_PIPELINE_SOURCE" = "merge_request_event" ]; then
      git fetch --quiet origin "$CI_MERGE_REQUEST_TARGET_BRANCH_NAME"
      GATE="$(mktemp -d)/gate"
      git worktree add --detach "$GATE" "$CI_MERGE_REQUEST_DIFF_BASE_SHA"
      (cd "$GATE" && pnpm install --frozen-lockfile && pnpm --filter @blackbox/verify build && pnpm --filter @blackbox/receipts build && pnpm --filter @blackbox/meterless build && pnpm --filter @blackbox/proxy build)
      BLACKBOX_CLI="$GATE/packages/proxy/dist/cli.js"
      SCRIPTS="$GATE/scripts/ci"
    fi
    export BLACKBOX_CLI SCRIPTS
